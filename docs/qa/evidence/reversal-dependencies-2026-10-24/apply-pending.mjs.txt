import assert from 'node:assert/strict';
import {readFile,readdir} from 'node:fs/promises';
import {createHash} from 'node:crypto';
import {Pool} from '@neondatabase/serverless';
import {assertIsolatedTargets} from '../../../tests/database/helpers/isolation.mjs';
import {resolveOwnerDatabaseUrl} from '../../../scripts/environment-config.mjs';
const owner=resolveOwnerDatabaseUrl(process.env);assertIsolatedTargets(owner,process.env.DATABASE_RUNTIME_URL);
const p=new Pool({connectionString:owner});const db=await p.connect();
const digest=text=>createHash('sha256').update(text).digest('hex');
const directory=new URL('../../../database/migrations/',import.meta.url);
const legacy=JSON.parse(await readFile(new URL('../../../scripts/migration-history-overrides.json',import.meta.url),'utf8'));
try {
 await db.query('BEGIN');await db.query('SELECT pg_advisory_xact_lock(1095580483::bigint)');
 const applied=new Map((await db.query('SELECT migration_name,sha256 FROM app_meta.schema_migrations')).rows.map(r=>[r.migration_name,r.sha256.trim()]));
 const files=(await readdir(directory)).filter(f=>/^\d{4}_[a-z0-9_]+\.sql$/.test(f)).sort(),pending=[];
 for(const name of files){const sql=(await readFile(new URL(name,directory),'utf8')).replaceAll('\r\n','\n'),hash=digest(sql);if(applied.has(name)){assert.ok(applied.get(name)===hash||applied.get(name)===legacy[name],`Applied migration changed: ${name}`);}else pending.push({name,sql,hash});}
 assert.deepEqual(pending.map(x=>x.name),['0125_dated_linked_settlement_reversal_dependencies.sql']);
 const facts=async()=> (await db.query(`SELECT md5(coalesce(string_agg(h,'' ORDER BY h),'')) AS fingerprint FROM (
 SELECT md5(to_jsonb(x)::text) AS h FROM finance.business_documents x UNION ALL SELECT md5(to_jsonb(x)::text) FROM finance.journal_entries x
 UNION ALL SELECT md5(to_jsonb(x)::text) FROM finance.journal_lines x UNION ALL SELECT md5(to_jsonb(x)::text) FROM finance.open_items x
 UNION ALL SELECT md5(to_jsonb(x)::text) FROM finance.settlement_allocations x UNION ALL SELECT md5(to_jsonb(x)::text) FROM finance.allocation_reversals x) rows`)).rows[0].fingerprint;
 const before=await facts();
 for(const file of pending){await db.query(file.sql);await db.query('INSERT INTO app_meta.schema_migrations(migration_name,sha256) VALUES($1,$2)',[file.name,file.hash]);}
 assert.equal(await facts(),before);await db.query('COMMIT');
 console.log(JSON.stringify({isolated:true,database:new URL(owner).pathname,previousReceipts:applied.size,applied:pending.map(x=>({name:x.name,sha256:x.hash})),unchangedFinancialFacts:true}));
} catch(error){await db.query('ROLLBACK').catch(()=>{});throw error;}finally{db.release();await p.end();}
